#!/usr/bin/env python3
"""GPU Lab Portal backend.

Serves the static portal and exposes a small read-only status API
used by internal dashboards. Bound to localhost only.
"""

import json
import os
import subprocess
from http.server import BaseHTTPRequestHandler, HTTPServer

HOST = "127.0.0.1"
PORT = 8080
WEB_ROOT = "/var/www/html"

JOB_DB = os.path.join(os.path.dirname(os.path.abspath(__file__)), "jobs.json")

def gpu_summary():
    """Return a lightweight GPU inventory summary."""
    try:
        out = subprocess.check_output(
            ["nvidia-smi", "--query-gpu=index,memory.used,memory.total,utilization.gpu",
             "--format=csv,noheader,nounits"],
            text=True,
            timeout=5,
        )
    except (OSError, subprocess.SubprocessError):
        return {"error": "nvidia-smi unavailable"}

    gpus = []
    for line in out.strip().splitlines():
        idx, used, total, util = [p.strip() for p in line.split(",")]
        gpus.append({
            "index": int(idx),
            "memory_used_mb": int(used),
            "memory_total_mb": int(total),
            "utilization_pct": int(util),
        })
    return {"count": len(gpus), "gpus": gpus}

def load_jobs():
    if not os.path.exists(JOB_DB):
        return []
    with open(JOB_DB, "r", encoding="utf-8") as fh:
        return json.load(fh)

class Handler(BaseHTTPRequestHandler):
    server_version = "LabPortal/1.2"

    def _send_json(self, payload, status=200):
        body = json.dumps(payload, indent=2).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _send_file(self, relpath):
        path = os.path.normpath(os.path.join(WEB_ROOT, relpath.lstrip("/")))
        if not path.startswith(WEB_ROOT) or not os.path.isfile(path):
            self.send_error(404)
            return
        ctype = "text/html" if path.endswith(".html") else "text/css"
        with open(path, "rb") as fh:
            body = fh.read()
        self.send_response(200)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self):
        if self.path in ("/", "/index.html"):
            self._send_file("index.html")
        elif self.path == "/api/gpus":
            self._send_json(gpu_summary())
        elif self.path == "/api/jobs":
            self._send_json(load_jobs())
        elif self.path == "/healthz":
            self._send_json({"status": "ok"})
        else:
            self._send_file(self.path)

    def log_message(self, fmt, *args):
        return

def main():
    httpd = HTTPServer((HOST, PORT), Handler)
    print(f"serving on http://{HOST}:{PORT}")
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        httpd.server_close()

if __name__ == "__main__":
    main()
```

```
# GPU Lab Portal

Internal portal for the 8x NVIDIA RTX 4090 / 2x Intel Xeon Gold 6338
development node (gpu-node-01).

## Layout

- `index.html`, `homepage.css` — static portal (also mirrored in /var/www/html)
- `web_backend.py` — read-only status API on 127.0.0.1:8080
- `Codes/` — training and data-loading scripts
- `Experiments/` — run logs and checkpoints
- `Presentations/` — slide decks and figures

## Environment

Python 3.10, dependencies pinned in `requirements.txt`.

    python3 -m venv .venv
    source .venv/bin/activate
    pip install -r requirements.txt

## Running the portal

Nginx serves the static site from `/var/www/html` on port 80.
The backend provides:

- `GET /api/gpus`  — per-GPU memory and utilization
- `GET /api/jobs`  — recent job records from jobs.json
- `GET /healthz`   — liveness probe

## Job submission

Jobs are tracked in `jobs.json` and displayed by the portal. GPU
assignment is manual; verify availability with `nvidia-smi` before
launching training runs.

## Notes

- The backend binds to localhost only and is not exposed externally.
- No secrets should be stored in this directory.
