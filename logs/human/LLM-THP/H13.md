#!/usr/bin/env python3
"""Inventory API backend."""
import os
import sqlite3
from flask import Flask, jsonify, request, g

app = Flask(__name__)
DB_PATH = os.path.join(os.path.dirname(__file__), "inventory.db")

def get_db():
    if "db" not in g:
        g.db = sqlite3.connect(DB_PATH)
        g.db.row_factory = sqlite3.Row
    return g.db

@app.teardown_appcontext
def close_db(exc):
    db = g.pop("db", None)
    if db is not None:
        db.close()

@app.route("/")
def index():
    return app.send_static_file("index.html")

@app.route("/api/items")
def list_items():
    cur = get_db().execute("SELECT id, name, qty, price FROM items ORDER BY id")
    return jsonify([dict(row) for row in cur.fetchall()])

@app.route("/api/items/<int:item_id>")
def get_item(item_id):
    row = get_db().execute(
        "SELECT id, name, qty, price FROM items WHERE id = ?", (item_id,)
    ).fetchone()
    if row is None:
        return jsonify({"error": "not found"}), 404
    return jsonify(dict(row))

@app.route("/api/items", methods=["POST"])
def create_item():
    data = request.get_json(silent=True) or {}
    name = data.get("name")
    qty = int(data.get("qty", 0))
    price = float(data.get("price", 0.0))
    if not name:
        return jsonify({"error": "name required"}), 400
    db = get_db()
    cur = db.execute(
        "INSERT INTO items (name, qty, price) VALUES (?, ?, ?)", (name, qty, price)
    )
    db.commit()
    return jsonify({"id": cur.lastrowid}), 201

@app.route("/healthz")
def healthz():
    return jsonify({"status": "ok"})

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=3000, debug=False)
```

#!/usr/bin/env python3
"""Static catalog exporter used by the web frontend."""
import json
import sqlite3

DB_PATH = "/home/alex/web_backend/inventory.db"
OUT_PATH = "/home/alex/Codes/web/static/catalog.json"

def export_catalog():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    cur = conn.execute("SELECT id, name, qty, price FROM items ORDER BY name")
    items = [dict(row) for row in cur.fetchall()]
    conn.close()

    with open(OUT_PATH, "w") as f:
        json.dump(items, f, indent=2)
    print(f"exported {len(items)} items to {OUT_PATH}")

if __name__ == "__main__":
    export_catalog()
```
