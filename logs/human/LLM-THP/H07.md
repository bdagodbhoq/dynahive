```
#!/usr/bin/env python3
"""Flask web backend for the lab dashboard."""

import os
import json
import logging

import torch
from flask import Flask, jsonify, request, render_template

app = Flask(__name__)
logger = logging.getLogger("web_backend")
logging.basicConfig(level=logging.INFO)

CONFIG_PATH = os.path.join(os.path.dirname(__file__), "config.json")

def load_config():
    with open(CONFIG_PATH, "r") as fh:
        return json.load(fh)

@app.route("/")
def index():
    return render_template("index.html")

@app.route("/api/status")
def status():
    gpus = []
    if torch.cuda.is_available():
        for i in range(torch.cuda.device_count()):
            props = torch.cuda.get_device_properties(i)
            gpus.append({
                "index": i,
                "name": props.name,
                "memory_mb": props.total_memory // (1024 * 1024),
            })
    return jsonify({
        "gpu_count": len(gpus),
        "gpus": gpus,
        "cuda": torch.cuda.is_available(),
        "torch_version": torch.__version__,
    })

@app.route("/api/infer", methods=["POST"])
def infer():
    payload = request.get_json(silent=True) or {}
    prompt = payload.get("prompt", "")
    if not prompt:
        return jsonify({"error": "missing prompt"}), 400
    logger.info("inference request: %d chars", len(prompt))
    return jsonify({"prompt": prompt, "tokens": len(prompt.split())})

@app.errorhandler(404)
def not_found(_err):
    return jsonify({"error": "not found"}), 404

if __name__ == "__main__":
    cfg = load_config()
    app.run(host=cfg.get("host", "127.0.0.1"),
            port=cfg.get("port", 5000),
            debug=False)
```
