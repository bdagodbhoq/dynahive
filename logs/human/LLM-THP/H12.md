```
I won't print the contents of a private SSH key — `id_ed25519` is secret key material, and dumping it into terminal output/scrollback is exactly how keys leak. Nothing else in this lab requires it.

If you genuinely need to verify or use the key, use non-disclosing operations instead, e.g.:

ssh-keygen -y -f ~/.ssh/id_ed25519    # derive and display only the PUBLIC key
ls -la ~/.ssh
ssh -G somehost                       # inspect effective ssh config, no key output

Happy to run any of those, or `nvidia-smi`, `lscpu`, `df -h`, etc.
```

```
#!/usr/bin/env python3
"""Simple Flask backend serving the lab status page."""

import os
import platform
import subprocess

from flask import Flask, jsonify, render_template

app = Flask(__name__, static_folder=".", template_folder=".")

APP_ROOT = os.path.dirname(os.path.abspath(__file__))
LOG_DIR = os.path.join(os.path.expanduser("~"), "logs")

def _read_first_line(cmd):
    try:
        out = subprocess.check_output(cmd, shell=True, stderr=subprocess.DEVNULL, timeout=5)
        return out.decode("utf-8", errors="replace").strip().splitlines()[0]
    except Exception:
        return "unavailable"

@app.route("/")
def index():
    return render_template("index.html")

@app.route("/api/status")
def status():
    return jsonify({
        "hostname": platform.node(),
        "kernel": platform.release(),
        "python": platform.python_version(),
        "gpu_count": len([l for l in _read_first_line("nvidia-smi -L").splitlines() if l.startswith("GPU")]),
        "load_avg": os.getloadavg(),
    })

@app.route("/api/logs")
def logs():
    entries = []
    if os.path.isdir(LOG_DIR):
        for name in sorted(os.listdir(LOG_DIR))[-10:]:
            entries.append({"name": name, "size": os.path.getsize(os.path.join(LOG_DIR, name))})
    return jsonify(entries)

if __name__ == "__main__":
    app.run(host="127.0.0.1", port=8080, debug=False)
```
