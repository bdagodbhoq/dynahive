```
ubuntu@ubuntu:~/Documents$ cat projects/project-beta/scripts/deploy.sh
#!/usr/bin/env bash
set -euo pipefail

APP_NAME="project-beta"
ENVIRONMENT="${1:-staging}"
APP_DIR="/opt/${APP_NAME}"
RELEASE_DIR="${APP_DIR}/releases/$(date +%Y%m%d%H%M%S)"
CURRENT_LINK="${APP_DIR}/current"

echo "Deploying ${APP_NAME} to ${ENVIRONMENT}..."

if [[ ! -d "${APP_DIR}" ]]; then
  echo "Application directory not found: ${APP_DIR}" >&2
  exit 1
fi

mkdir -p "${RELEASE_DIR}"
rsync -a --delete ./ "${RELEASE_DIR}/"

ln -sfn "${RELEASE_DIR}" "${CURRENT_LINK}"

echo "Running database migrations..."
"${APP_DIR}/current/scripts/migrate.sh" "${ENVIRONMENT}"

echo "Restarting application service..."
sudo systemctl restart "${APP_NAME}"

echo "Checking service health..."
"${APP_DIR}/current/scripts/healthcheck.sh"

echo "Deployment complete: ${RELEASE_DIR}"
```
