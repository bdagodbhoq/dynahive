`ubuntu@ubuntu:~$ cat projects/webapp/index.js`

```
const express = require('express');
const path = require('path');
const fs = require('fs');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(express.json());
app.use(express.static(path.join(__dirname, 'public')));

// Request log
app.use((req, res, next) => {
  console.log(`${new Date().toISOString()} ${req.method} ${req.url}`);
  next();
});

// Health check
app.get('/health', (req, res) => {
  res.json({ status: 'ok', uptime: process.uptime() });
});

// Simple in-memory items API
let items = [
  { id: 1, name: 'widget', qty: 12 },
  { id: 2, name: 'gadget', qty: 5 },
];

app.get('/api/items', (req, res) => {
  res.json(items);
});

app.get('/api/items/:id', (req, res) => {
  const item = items.find((i) => i.id === Number(req.params.id));
  if (!item) return res.status(404).json({ error: 'not found' });
  res.json(item);
});

app.post('/api/items', (req, res) => {
  const { name, qty } = req.body || {};
  if (!name) return res.status(400).json({ error: 'name required' });
  const id = items.length ? Math.max(...items.map((i) => i.id)) + 1 : 1;
  const item = { id, name, qty: qty || 0 };
  items.push(item);
  res.status(201).json(item);
});

app.listen(PORT, () => {
  console.log(`webapp listening on http://localhost:${PORT}`);
});

module.exports = app;
```

`ubuntu@ubuntu:~$ cat projects/webapp/README.md`

```
# webapp

A small Express.js service used for the internal dashboard prototypes. It serves
static assets from `public/` and exposes a JSON items API.

## Requirements

- Node.js 18+
- npm 9+

## Getting started

npm install
npm start

The server listens on `PORT` (default `3000`).

## Scripts

| Command | Description |
| --- | --- |
| `npm start` | Start the server |
| `npm run dev` | Start with nodemon (auto-reload) |
| `npm test` | Run the test suite |

## Endpoints

| Method | Path | Description |
| --- | --- | --- |
| GET | `/health` | Liveness probe, returns `{ "status": "ok", "uptime": <seconds> }` |
| GET | `/api/items` | List all items |
| GET | `/api/items/:id` | Fetch a single item by id |

Items are held in memory and reset whenever the process restarts, so this is
fine for local work but not for anything persistent.

## Layout

index.js        server entry point
public/         static files served at /
package.json    dependencies and scripts

## Notes

- Request logging goes to stdout: `<ISO timestamp> <METHOD> <url>`.
- `node_modules/` and `.env` are git-ignored; don't commit them.
- This is a prototype — no auth layer, no database, no rate limiting.
```
