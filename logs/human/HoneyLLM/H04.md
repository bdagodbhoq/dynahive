`ubuntu@ubuntu:~$ cat projects/backup/backup.sh`

```
#!/bin/bash
# backup.sh - rsync project files to the NAS and rotate oversized logs
# created 2025-02-13

set -euo pipefail

SRC="/home/ubuntu/projects/"
DEST="/mnt/nas/backup/"
NAS="192.168.1.100:/volume1/backup"
LOG="/var/log/backup.log"

log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG"
}

if ! mountpoint -q /mnt/nas; then
    log "NAS not mounted, attempting mount"
    sudo mount -t nfs "$NAS" /mnt/nas || { log "mount failed, aborting"; exit 1; }
fi

log "starting backup: $SRC -> $DEST"
rsync -av --delete "$SRC" "$DEST" >>"$LOG" 2>&1
log "rsync finished"

log "truncating logs larger than 100M"
sudo find /var/log -type f -name "*.log" -size +100M -exec truncate -s 0 {} \;

log "backup complete"
```
