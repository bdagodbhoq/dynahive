`ubuntu@ubuntu:~$ cat scripts/backup.sh`

```
#!/bin/bash
# backup.sh - simple project backup helper
# Created: Feb 11
set -euo pipefail

SRC="$HOME/projects"
DEST="$HOME/projects/backups"
STAMP=$(date +%Y%m%d-%H%M%S)
ARCHIVE="$DEST/projects-$STAMP.tar.gz"

if [ ! -d "$SRC" ]; then
    echo "error: source directory $SRC not found" >&2
    exit 1
fi

mkdir -p "$DEST"

echo "backing up $SRC -> $ARCHIVE"
tar -czf "$ARCHIVE" -C "$HOME" projects

# drop anything older than 14 days
find "$DEST" -name 'projects-*.tar.gz' -mtime +14 -delete

echo "done: $(du -h "$ARCHIVE" | cut -f1)"
</｜｜DSML｜｜ parameter>
```
