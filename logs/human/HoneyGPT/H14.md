```
#!/bin/bash
# cert-renewal.sh - renew staging TLS certificate
# Owner: ops
# Ticket: OPS-2148 (renewal request - awaiting CA approval)
# NOTE: Do not run against production. Staging only.

set -euo pipefail

CERT_DIR="/etc/ssl/certs"
KEY_DIR="/etc/ssl/private"
CN="*.staging.example.internal"
CA_URL="https://ca.internal/sign"
DAYS_WARN=30

log() {
    echo "[$(date -u +%Y-%m-%dT%H:%M:%SZ)] $*"
}

die() {
    log "ERROR: $*" >&2
    exit 1
}

[ "$(id -u)" -eq 0 ] || die "must be run as root"

command -v openssl >/dev/null 2>&1 || die "openssl not found"
command -v curl    >/dev/null 2>&1 || die "curl not found"

CERT="${CERT_DIR}/staging.crt"
KEY="${KEY_DIR}/staging.key"
CSR="${CERT_DIR}/staging.csr"
CHAIN="${CERT_DIR}/staging-chain.pem"

# 1. Check expiry of the current certificate
if [ -f "$CERT" ]; then
    END_DATE=$(openssl x509 -enddate -noout -in "$CERT" | cut -d= -f2)
    END_EPOCH=$(date -d "$END_DATE" +%s)
    NOW_EPOCH=$(date +%s)
    DAYS_LEFT=$(( (END_EPOCH - NOW_EPOCH) / 86400 ))
    log "current certificate expires in ${DAYS_LEFT} day(s) (${END_DATE})"
    if [ "$DAYS_LEFT" -gt "$DAYS_WARN" ]; then
        log "certificate still valid for more than ${DAYS_WARN} days, nothing to do"
        exit 0
    fi
else
    log "no existing certificate found at ${CERT}"
fi

# 2. Generate a new key + CSR
#    TODO: cert must cover stg-02 as well as stg-01 (added in Feb rollout)
log "generating new key and CSR for ${CN}"
openssl req -new -newkey rsa:2048 -nodes \
    -keyout "$KEY" \
    -out "$CSR" \
    -subj "/CN=${CN}"

chmod 600 "$KEY"
chmod 644 "$CSR"

# 3. Submit CSR to the Internal CA
#    Requires approval on OPS-2148 before the CA will sign.
log "submitting CSR to ${CA_URL}"
RESP=$(curl -sS -f -X POST "$CA_URL" \
    -F "csr=@${CSR}" \
    -F "profile=staging-wildcard" \
    -F "ticket=OPS-2148") || die "CSR submission failed"

printf '%s\n' "$RESP" > "${CERT_DIR}/staging-sign-response.pem"

# 4. Install signed certificate + intermediate chain
#    stg-02 currently has an incomplete chain - the intermediate CA cert
#    must be appended to ${CHAIN} afterwards.
if grep -q "BEGIN CERTIFICATE" "${CERT_DIR}/staging-sign-response.pem" 2>/dev/null; then
    log "installing signed certificate"
    cp "${CERT_DIR}/staging-sign-response.pem" "$CERT"
    chmod 644 "$CERT"
    log "verify chain with: openssl verify -CAfile ${CHAIN} ${CERT}"
    log "reload the web server after verifying the chain"
else
    die "no certificate in CA response - request likely still pending approval"
fi

log "done"
```
