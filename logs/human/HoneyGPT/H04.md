```
#!/bin/bash
# deploy.sh - simple deploy helper for internal apps
set -euo pipefail

APP_DIR="/var/www/html"
BACKUP_DIR="/var/backups/deploy"
STAMP=$(date +%Y%m%d-%H%M%S)

echo "[*] Starting deploy at ${STAMP}"

if [ ! -d "${APP_DIR}" ]; then
    echo "[!] ${APP_DIR} not found" >&2
    exit 1
fi

mkdir -p "${BACKUP_DIR}"
tar -czf "${BACKUP_DIR}/html-${STAMP}.tar.gz" -C "${APP_DIR}" . \
    && echo "[+] Backup written to ${BACKUP_DIR}/html-${STAMP}.tar.gz"

cd "${APP_DIR}"
git pull --ff-only origin main

sudo systemctl reload nginx
echo "[+] Deploy complete"
```

```
#!/bin/bash
# backup.sh - nightly backup of /var/www
set -euo pipefail

SRC_DIR="/var/www"
DEST_DIR="/var/backups/www"
STAMP=$(date +%Y%m%d-%H%M%S)
KEEP_DAYS=7

echo "[*] Backup started at ${STAMP}"

if [ ! -d "${SRC_DIR}" ]; then
    echo "[!] Source ${SRC_DIR} does not exist" >&2
    exit 1
fi

mkdir -p "${DEST_DIR}"
tar -czf "${DEST_DIR}/www-${STAMP}.tar.gz" -C "${SRC_DIR}" . \
    && echo "[+] Archive: ${DEST_DIR}/www-${STAMP}.tar.gz"

find "${DEST_DIR}" -name 'www-*.tar.gz' -type f -mtime +${KEEP_DAYS} -delete
echo "[*] Pruned archives older than ${KEEP_DAYS} days"

echo "[+] Backup complete"
```

```
from flask import Blueprint, jsonify, request

from .models import Item
from .utils import parse_pagination

bp = Blueprint("api", __name__)

@bp.route("/health", methods=["GET"])
def health():
    return jsonify({"status": "ok"}), 200

@bp.route("/items", methods=["GET"])
def list_items():
    page, per_page = parse_pagination(request.args)
    items = Item.query.order_by(Item.id).paginate(page, per_page, False)
    return jsonify({
        "items": [i.to_dict() for i in items.items],
        "page": items.page,
        "pages": items.pages,
        "total": items.total,
    })

@bp.route("/items/<int:item_id>", methods=["GET"])
def get_item(item_id):
    item = Item.query.get(item_id)
    if item is None:
        return jsonify({"error": "not found"}), 404
    return jsonify(item.to_dict())

@bp.route("/items", methods=["POST"])
def create_item():
    payload = request.get_json(silent=True) or {}
    name = payload.get("name")
    if not name:
        return jsonify({"error": "name is required"}), 400
    item = Item(name=name)
    item.save()
    return jsonify(item.to_dict()), 201
```
