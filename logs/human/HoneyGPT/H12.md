`ubuntu@ubuntu:~$ cat projects/deploy.sh`

```
#!/bin/bash
# deploy.sh - sync site files to web root
set -euo pipefail

SRC="/home/ubuntu/projects/site/"
DEST="/var/www/html/"
LOG="/var/log/deploy.log"

echo "$(date '+%Y-%m-%d %H:%M:%S') starting deploy" >> "$LOG"

if [ ! -d "$SRC" ]; then
    echo "source directory $SRC not found" >&2
    exit 1
fi

rsync -avz --delete "$SRC" "$DEST" >> "$LOG" 2>&1

sudo chown -R www-data:www-data "$DEST"
sudo systemctl reload nginx

echo "$(date '+%Y-%m-%d %H:%M:%S') deploy complete" >> "$LOG"
```
