```
#!/usr/bin/env bash
set -euo pipefail

APP_NAME="api-service"
REMOTE_USER="deploy"
REMOTE_HOST="staging.example.internal"
REMOTE_DIR="/srv/${APP_NAME}"
SSH_KEY="${HOME}/.ssh/id_staging"

echo "Deploying ${APP_NAME} to ${REMOTE_HOST}..."

cd "$(dirname "$0")/../api-service"
npm ci
npm test
npm run build

rsync -avz --delete -e "ssh -i ${SSH_KEY}" \
  ./dist/ "${REMOTE_USER}@${REMOTE_HOST}:${REMOTE_DIR}/"

ssh -i "${SSH_KEY}" "${REMOTE_USER}@${REMOTE_HOST}" \
  "cd ${REMOTE_DIR} && npm ci --production && sudo systemctl restart ${APP_NAME}"

echo "Deploy complete."
```

`ubuntu@ubuntu:~/projects/scripts$ cat backup.sh`

```
#!/usr/bin/env bash
set -euo pipefail

BACKUP_DIR="/tmp/backups"
SRC_DIR="${HOME}/projects"
STAMP="$(date +%Y%m%d-%H%M%S)"
ARCHIVE="${BACKUP_DIR}/projects-${STAMP}.tar.gz"

mkdir -p "${BACKUP_DIR}"

tar -czf "${ARCHIVE}" -C "$(dirname "${SRC_DIR}")" "$(basename "${SRC_DIR}")"

find "${BACKUP_DIR}" -type f -name 'projects-*.tar.gz' -mtime +7 -delete

echo "Backup written to ${ARCHIVE}"
```
