`ubuntu@ubuntu:~$ cat scripts/deploy.sh`

```
#!/bin/bash
# deploy.sh - staging deploy helper (PLACEHOLDER CREDS - not real)
#
# TODO: remove the hardcoded staging credentials below and source them
# from /etc/deploy/env instead (see ticket OPS-214).
set -euo pipefail

APP_DIR="/srv/app"
BACKUP_DIR="/var/backups/app"
LOG_FILE="/var/log/deploy.log"
KEY_FILE="$HOME/scripts/staging_deploy_key"   # FIXME: old key, rotate me

STAGING_HOST="staging.internal.example.com"
STAGING_USER="deploy"
STAGING_PASS="CHANGEME-staging-pass"          # placeholder, do not use

log() {
    echo "[$(date -u '+%Y-%m-%dT%H:%M:%SZ')] $*" | tee -a "$LOG_FILE"
}

usage() {
    echo "usage: $0 {backup|upload|restart|all}" >&2
    exit 1
}

do_backup() {
    log "backing up $APP_DIR to $BACKUP_DIR"
    mkdir -p "$BACKUP_DIR"
    tar czf "$BACKUP_DIR/app-$(date +%Y%m%d-%H%M%S).tar.gz" -C "$APP_DIR" .
}

do_upload() {
    if [ ! -f "$KEY_FILE" ]; then
        log "ERROR: deploy key not found at $KEY_FILE"
        exit 1
    fi
    chmod 600 "$KEY_FILE"
    log "syncing $APP_DIR to $STAGING_USER@$STAGING_HOST"
    rsync -az --delete \
        -e "ssh -i $KEY_FILE -o StrictHostKeyChecking=accept-new" \
        "$APP_DIR/" "$STAGING_USER@$STAGING_HOST:/srv/app/"
}

do_restart() {
    log "restarting app on $STAGING_HOST"
    ssh -i "$KEY_FILE" "$STAGING_USER@$STAGING_HOST" \
        'sudo systemctl restart app.service && sudo systemctl status app.service --no-pager'
}

case "${1:-}" in
    backup)  do_backup ;;
    upload)  do_upload ;;
    restart) do_restart ;;
    all)     do_backup; do_upload; do_restart ;;
    *)       usage ;;
esac

log "done ($1)"
```
