```
#!/bin/bash
# staging-api startup script
# Managed by systemd: /etc/systemd/system/app.service

set -euo pipefail

APP_NAME="staging-api"
APP_DIR="/srv/app/staging"
VENV_DIR="${APP_DIR}/venv"
LOG_DIR="/var/log/app"
PID_FILE="/run/${APP_NAME}.pid"

export APP_ENV="staging"
export CONFIG_PATH="${APP_DIR}/config.yml"

log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*"
}

if [ ! -d "${APP_DIR}" ]; then
    log "ERROR: application directory ${APP_DIR} not found"
    exit 1
fi

if [ ! -x "${VENV_DIR}/bin/python" ]; then
    log "ERROR: virtualenv not found at ${VENV_DIR}"
    exit 1
fi

mkdir -p "${LOG_DIR}"

# Load DB_PASSWORD from environment file if not already set
if [ -z "${DB_PASSWORD:-}" ] && [ -f "/etc/${APP_NAME}/env" ]; then
    log "Loading environment from /etc/${APP_NAME}/env"
    set -a
    . "/etc/${APP_NAME}/env"
    set +a
fi

if [ -z "${DB_PASSWORD:-}" ]; then
    log "ERROR: DB_PASSWORD is not set"
    exit 1
fi

cd "${APP_DIR}"

log "Starting ${APP_NAME} in ${APP_ENV}"
exec "${VENV_DIR}/bin/gunicorn" \
    --bind 0.0.0.0:8080 \
    --workers 4 \
    --pid "${PID_FILE}" \
    --access-logfile "${LOG_DIR}/app.log" \
    --error-logfile "${LOG_DIR}/app.log" \
    --log-level info \
    wsgi:application
```
